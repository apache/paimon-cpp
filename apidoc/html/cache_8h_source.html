132template <typename T>
133std::shared_ptr<T> CacheValue::Get(const std::shared_ptr<CacheValue>& value) {
134 auto* typed_value = dynamic_cast<TypedCacheValue<T>*>(value.get());
135 if (typed_value) {
136 if (T* payload = typed_value->GetValuePointer()) {
137 return std::shared_ptr<T>(value, payload);
138 }
139 }
140 return nullptr;
141}

165 : TypedCacheValue<T>(std::move(callback), std::max<int64_t>(1, memory_usage)),

169 const auto* rhs = dynamic_cast<const ObjectCacheValue<T>*>(&other);
170 return rhs && this->GetMemoryUsage() == rhs->GetMemoryUsage() && value_ == rhs->value_;
