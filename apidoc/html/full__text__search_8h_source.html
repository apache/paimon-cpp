47 std::shared_ptr<FullTextSearch> OffsetRange(int64_t from, int64_t to) const {
48 if (!include_row_ids) {
49 return std::make_shared<FullTextSearch>(*this);
50 }
51 const RoaringBitmap64& global_row_ids = include_row_ids.value();
52 RoaringBitmap64 local_row_ids;
53 auto iter = global_row_ids.EqualOrLarger(from);
54 auto end = global_row_ids.End();
55 if (iter != end && *iter <= to) {
56 // Shift one run of consecutive row ids as a range, so that converting a range of rows
57 // to rank does not visit each row.
58 int64_t first = *global_row_ids.Begin();
59 int64_t last = first + (global_row_ids.Cardinality() - 1);
60 if (last == RoaringBitmap64::MAX_VALUE ||
61 global_row_ids.EqualOrLarger(last + 1) == end) {
62 int64_t local_to = std::min(last, to) - from;
63 local_row_ids.AddRange(*iter - from, local_to);
64 local_row_ids.Add(local_to);
65 } else {
66 for (; iter != end && *iter <= to; ++iter) {
67 local_row_ids.Add(*iter - from);
68 }
69 }
70 }
71 return std::make_shared<FullTextSearch>(field_name, query, limit, std::move(local_row_ids));
72 }
