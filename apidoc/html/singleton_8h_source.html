31class PAIMON_EXPORT LazyInstantiation {
32 protected:
33 template <typename T>
34 static void Create(T*& ptr) {
35 // Publication ordering is handled by the release store in
36 // Singleton<T, InstPolicy>::GetInstance(), so no barrier is needed here.
37 ptr = new T;
38 static std::shared_ptr<T> destroyer(ptr);
39 }
40};

61template <typename T, typename InstPolicy>
62T* Singleton<T, InstPolicy>::GetInstance() {
63 static std::atomic<T*> ptr{nullptr};
64 static std::mutex mutex;
65 T* p = ptr.load(std::memory_order_acquire);
66 if (PAIMON_UNLIKELY(p == nullptr)) {
67 std::lock_guard<std::mutex> lg(mutex);
68 // Re-check under the mutex with a relaxed load; the mutex already
69 // synchronizes with the creating thread.
70 p = ptr.load(std::memory_order_relaxed);
71 if (p == nullptr) {
72 InstPolicy::Create(p);
73 ptr.store(p, std::memory_order_release);
74 }
75 }
76 return p;
77}

79// FactoryCreator and IOHook are instantiated exactly once in singleton.cpp, and the
80// extern declarations below suppress implicit instantiation everywhere else. The
81// file-format/file-system plugins are separate shared libraries linked with
82// -Bsymbolic, so a per-library copy of GetInstance()'s function-local static state
83// would never be interposed: factory registrations would land in a different
84// instance than lookups. Do not replace these with implicit instantiation. Types local to a single
85// translation unit (e.g. test-only types) can still instantiate Singleton<T>
86// implicitly because they cannot span library boundaries.
87class FactoryCreator;
88class IOHook;
