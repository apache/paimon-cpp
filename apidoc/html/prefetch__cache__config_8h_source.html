94 // The defaults are aligned with the reader's request granularity and with
95 // realistic data file sizes:
96 // - range_size_limit matches the parquet reader's 32 MiB request blocks
97 // (Arrow ReadRangeCache's own range limit); a smaller limit cuts entries
98 // below the request size, so a request can never be served from one piece.
99 // - pre_buffer_limit must exceed the LARGEST single read a reader issues
100 // (coalesced column-chunk reads of ~128 MiB were observed): fetches are
101 // only dispatched up to this window, so a request reaching past it can
102 // never be served and falls back to a second fetch of the same bytes.
103 uint64_t range_size_limit_ = 32 * 1024 * 1024;
104 uint64_t hole_size_limit_ = 8 * 1024;
105 uint64_t pre_buffer_limit_ = 256 * 1024 * 1024;
106 // Blocks are aligned to the END of the file, so a block never reaches past
107 // EOF. 64 KiB is the granularity the reads no prefetched range covers are
108 // shared at: small enough that a metadata read at the tail of a file is
109 // served by one block instead of straddling two, large enough that a block
110 // fetch does not pull in much more than the reads ask for.
111 uint64_t block_size_ = 64 * 1024;
112 // One block is enough for the metadata tail of a file; the limit only
113 // bounds the pathological case, as blocks are never evicted.
114 uint64_t block_cache_limit_ = 1024 * 1024;
115};
